## 문제 2: 리팩터링 실습 (난이도: ★★☆)

다음 코드를 리팩터링하시오 (매직 넘버 제거, 중복 함수화, 이름 개선):

```c
#include <stdio.h>

int main(void) {
    int a[10] = {3,7,1,9,2,8,4,6,5,10};
    int s = 0;
    for (int i = 0; i < 10; i++) s += a[i];
    printf("sum=%d avg=%.1f\n", s, (double)s/10);

    int m = a[0];
    for (int i = 1; i < 10; i++) {
        if (a[i] > m) m = a[i];
    }
    printf("max=%d\n", m);

    int mn = a[0];
    for (int i = 1; i < 10; i++) {
        if (a[i] < mn) mn = a[i];
    }
    printf("min=%d\n", mn);
    return 0;
}
```
